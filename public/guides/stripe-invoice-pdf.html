Branded Stripe receipt PDFs from invoice.paid

When Stripe marks an invoice paid, render your own receipt PDF and email it to the customer: one webhook handler, one DocForge call, one email.

Stripe already hosts an invoice_pdf for every invoice. Use this recipe when you want your own layout and branding, a different locale or date format, or the same receipt design for payments that do not go through Stripe Billing.

How it fits together

  1. Stripe sends invoice.paid to your endpoint. Verify the signature.
  2. Map the invoice object to the gallery:receipt fields (table below). Amounts stay in cents.
  3. POST /v1/render with "output": "base64".
  4. Attach the PDF to an email (Resend in this example; any provider with attachments works).
  5. Return 2xx only after the email is sent, so Stripe retries on failure. Deduplicate on event.id, because Stripe can deliver an event more than once.

Field mapping

Stripe invoicegallery:receiptNotes
numbernumbere.g. A1B2C3D4-0007
status_transitions.paid_atpaid_atEpoch seconds; the date helper detects that.
customer_emailcustomer_emailAlso the email recipient.
currencycurrencyLower-case is fine.
amount_paidamount_paidCents, passed through unchanged.
lines.data[].description / amount / quantitylines[].description / amount / quantityamount is the line total in cents. The payload holds the first page of lines; when lines.has_more is true, list the rest with the Stripe API.
(your settings)merchant, logo_url, support_email, localeConstant per account.
(optional)payment_methodCard brand and last 4 live on the payment, not the invoice, and where they sit depends on your Stripe API version. Look them up, or send a plain label such as "Card".

The webhook handler (Node + Express)

server.mjs
// npm i express stripe
import express from 'express';
import Stripe from 'stripe';

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY);
const app = express();
const done = new Set(); // use a database table in production

function receiptData(invoice, lines) {
  return {
    merchant: 'Bookly',
    logo_url: 'https://bookly.app/logo.png',
    number: invoice.number,
    paid_at: invoice.status_transitions?.paid_at ?? Math.floor(Date.now() / 1000),
    customer_email: invoice.customer_email,
    currency: invoice.currency,
    locale: 'en-US',
    amount_paid: invoice.amount_paid,
    lines: lines.map((l) => ({ description: l.description, amount: l.amount, quantity: l.quantity ?? 1 })),
    payment_method: 'Card',
    support_email: 'help@bookly.app',
  };
}

async function renderReceipt(data) {
  const res = await fetch('https://YOUR-DOMAIN/v1/render', {
    method: 'POST',
    headers: { Authorization: `Bearer ${process.env.DOCFORGE_KEY}`, 'Content-Type': 'application/json' },
    body: JSON.stringify({ template_id: 'gallery:receipt', output: 'base64', filename: `receipt-${data.number}.pdf`, data }),
    signal: AbortSignal.timeout(40_000),
  });
  const body = await res.json();
  if (!res.ok) throw new Error(`DocForge ${res.status} ${body.error}: ${body.message ?? ''}`);
  return body; // { filename, content_type, size, base64 }
}

async function emailReceipt(to, number, pdf) {
  const res = await fetch('https://api.resend.com/emails', {
    method: 'POST',
    headers: { Authorization: `Bearer ${process.env.RESEND_API_KEY}`, 'Content-Type': 'application/json' },
    body: JSON.stringify({
      from: 'Bookly <receipts@bookly.app>',
      to,
      subject: `Your receipt ${number}`,
      text: 'Thanks for your payment. Your receipt is attached.',
      attachments: [{ filename: pdf.filename, content: pdf.base64 }],
    }),
  });
  if (!res.ok) throw new Error(`Resend ${res.status}: ${await res.text()}`);
}

// Stripe signs the raw body, so this route must not use express.json().
app.post('/webhooks/stripe', express.raw({ type: 'application/json' }), async (req, res) => {
  let event;
  try {
    event = stripe.webhooks.constructEvent(req.body, req.get('stripe-signature'), process.env.STRIPE_WEBHOOK_SECRET);
  } catch (e) {
    return res.status(400).send(`Invalid signature: ${e.message}`);
  }
  if (event.type !== 'invoice.paid' || done.has(event.id)) return res.sendStatus(200);

  const invoice = event.data.object;
  if (!invoice.customer_email || invoice.amount_paid === 0) return res.sendStatus(200); // nothing to send
  try {
    let lines = invoice.lines.data;
    if (invoice.lines.has_more) {
      lines = [];
      for await (const line of stripe.invoices.listLineItems(invoice.id, { limit: 100 })) lines.push(line);
    }
    const pdf = await renderReceipt(receiptData(invoice, lines));
    await emailReceipt(invoice.customer_email, invoice.number, pdf);
    done.add(event.id);
    res.sendStatus(200);
  } catch (e) {
    console.error('receipt failed', event.id, e.message);
    res.sendStatus(500); // Stripe retries with backoff
  }
});

app.listen(3000);

What can go wrong

ResponseWhat to do
402 quota_exceededYour monthly renders and the 10% grace band (paid plans) are used up. The body carries upgrade_url. Returning 500 makes Stripe retry, so receipts go out once you upgrade.
422 template_errorA field the template needs is malformed. The message says which; fix the mapping.
429 concurrency_limit, 503 busyTemporary. Stripe's retry covers it, or wait for Retry-After and retry yourself.
X-Quota-Warning: over_limit headerYou are inside the grace band. Renders still succeed; time to upgrade.
